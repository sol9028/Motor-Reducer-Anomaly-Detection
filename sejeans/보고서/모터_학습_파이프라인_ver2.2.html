모터-감속기 이상 탐지 — 학습 파이프라인 (Stage 2~4)¶

본격 학습 파이프라인의 학습부 단독 노트북 · Google Colab 용

Stage 1 에서 만든 4피처 시계열 CSV 를 입력으로, (차종 × 센서) 슬롯별 모델을 학습합니다.

슬롯 라우팅 → chunker(2000) → 클래스 캡 → MiniRocket 1회 변환 → StratifiedGroupKFold(폴드마다 LR만 재학습) → ModelRegistry

사용법 — ① 환경 설정 셀에서 sktime 설치 + Drive 마운트 → ② 경로 & 파라미터 셀에서 CSV_ROOT / MODEL_DIR 두 곳만 채우기 → ③ 위에서 아래로 실행.

산출물: model_registry.joblib(슬롯별 모델), eval_results.json(평가).

2. 데이터 유틸 — 스캔 · 슬롯 분류 · 청킹¶

CSV 를 (vehicle, sensor) 슬롯으로 묶고, 각 CSV(연속 세션)를 CHUNK_SIZE(2000) 행씩 잘라 고정 길이 다변량 샘플(청크)로 만듭니다. 청크 1개는 원본 PNG 약 1장 분량이며, 각 청크는 원본 CSV 의 group_key 를 상속해 데이터 누수 방지에 사용됩니다. (슬롯·클래스당 청크 수 제한은 다음 단계의 CHUNK_CAP_PER_CLASS 에서 적용됩니다.)

# =====================================================================
#  데이터 유틸 — CSV 스캔 · 슬롯 분류 · 청킹
#   슬롯 = (vehicle, sensor)  →  최대 3차종 × 3센서 = 9개
#   청크 = 한 CSV(연속 세션)를 CHUNK_SIZE 행씩 자른 고정 길이 샘플
# =====================================================================
import os, glob
import numpy as np
import pandas as pd
from collections import defaultdict
from pathlib import Path


def find_csv_paths(root):
    return sorted(glob.glob(os.path.join(root, "**", "*.csv"), recursive=True))


def slot_of_csv(path):
    """CSV 첫 행에서 (vehicle, sensor, fault_class) 읽기 (가벼움)."""
    head = pd.read_csv(path, nrows=1, usecols=["vehicle", "sensor", "fault_class"])
    return (str(head["vehicle"].iloc[0]),
            str(head["sensor"].iloc[0]),
            str(head["fault_class"].iloc[0]))


def scan_slots(root):
    """모든 CSV 를 (vehicle, sensor) 슬롯으로 묶고, 슬롯별 등장 클래스 집합을 기록."""
    slots, slot_classes = defaultdict(list), defaultdict(set)
    for p in find_csv_paths(root):
        try:
            v, s, fc = slot_of_csv(p)
        except Exception as e:
            print("스킵(헤더 읽기 실패):", p, "→", e)
            continue
        slots[(v, s)].append(p)
        slot_classes[(v, s)].add(fc)
    return slots, slot_classes


def load_slot_chunks(paths):
    """
    슬롯에 속한 CSV 들을 읽어 청크 단위 샘플로 변환.
    반환: X (n_chunks, C, CHUNK_SIZE), y (n_chunks,), groups (n_chunks,) | None
    각 청크의 label = 해당 CSV 의 fault_class, group = group_key (누수 방지용).
    """
    Xs, ys, gs = [], [], []
    for p in paths:
        try:
            df = pd.read_csv(p)
        except Exception as e:
            print("스킵(읽기 실패):", p, "→", e); continue
        if df.empty:
            continue
        feats = df[FEATURE_COLS].to_numpy(dtype=np.float32)     # (n, C)
        n = (len(feats) // CHUNK_SIZE) * CHUNK_SIZE             # 끝의 잔여행(<2000) 버림
        if n == 0:
            continue
        C = feats.shape[1]
        chunks = feats[:n].T.reshape(C, -1, CHUNK_SIZE).transpose(1, 0, 2)  # (n_chunks, C, L)
        label = str(df["fault_class"].iloc[0])
        group = str(df["group_key"].iloc[0])
        Xs.append(chunks)
        ys += [label] * len(chunks)
        gs += [group] * len(chunks)
    if not Xs:
        return None
    return np.concatenate(Xs, 0), np.array(ys), np.array(gs)

4. 모델 정의 — MiniRocket(다변량) + LogisticRegression¶

채널별 표준화 → MiniRocket 특징 변환(슬롯당 1회) → StandardScaler → LogisticRegression.

무거운 MiniRocket 변환은 슬롯마다 한 번만 수행하고, 교차검증 폴드에서는 가벼운 LogisticRegression 만 다시 학습해 속도를 크게 높였습니다. 학습 전 CHUNK_CAP_PER_CLASS 로 슬롯·클래스당 청크 수를 제한해 중복을 줄입니다. 예측은 항상 5클래스(CLASS_ORDER) 정렬 확률을 반환하며(없는 클래스는 0), StratifiedGroupKFold 로 group 누수 방지 + 클래스 비율 유지를 동시에 달성하고, 마지막에 전체 데이터로 LR 을 재학습한 모델을 저장합니다.

# =====================================================================
#  모델 정의 — MiniRocket(다변량) + LogisticRegression
#   핵심 최적화:
#    (1) 슬롯·클래스당 청크 수를 CHUNK_CAP_PER_CLASS 로 제한 (중복 제거 → 속도/메모리↓)
#    (2) MiniRocket 변환을 "슬롯당 1번"만 하고 폴드마다 재사용 (변환 약 4배 절감)
#    (3) LR 반복수 축소. 5클래스 정렬 확률 반환은 그대로 유지
# =====================================================================
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LogisticRegression


def make_minirocket(n_channels):
    if n_channels == 1:
        from sktime.transformations.panel.rocket import MiniRocket
        return MiniRocket(num_kernels=NUM_KERNELS, random_state=RANDOM_STATE,
                          n_jobs=MINIROCKET_N_JOBS)
    from sktime.transformations.panel.rocket import MiniRocketMultivariate
    return MiniRocketMultivariate(num_kernels=NUM_KERNELS, random_state=RANDOM_STATE,
                                  n_jobs=MINIROCKET_N_JOBS)


def _fit_channel_stats(X):
    mean = X.mean(axis=(0, 2)).astype(np.float32)
    std  = X.std(axis=(0, 2)).astype(np.float32)
    std[std == 0] = 1.0
    return mean, std


def _apply_channel(X, mean, std):
    return ((X - mean[None, :, None]) / std[None, :, None]).astype(np.float32)


def cap_per_class(X, y, g, cap, seed=RANDOM_STATE):
    """슬롯·클래스당 청크 수를 cap 개로 제한(무작위 추출). cap 이 None 이면 그대로 둠.
    같은 세션 청크는 신호가 거의 같아 전부 쓸 필요가 없으므로 속도/메모리를 크게 줄인다."""
    if not cap:
        return X, y, g
    rng = np.random.default_rng(seed)
    keep = []
    for c in np.unique(y):
        idx = np.where(y == c)[0]
        keep.append(rng.choice(idx, min(cap, len(idx)), replace=False))
    keep = np.sort(np.concatenate(keep))
    return X[keep], y[keep], g[keep]


def build_features(X):
    """채널 표준화 + MiniRocket 적합/변환을 '한 번'만 수행.
    반환한 F(특징 행렬)를 교차검증 폴드와 최종 학습에서 공유해 변환 비용을 줄인다."""
    mean, std = _fit_channel_stats(X)
    Xn = _apply_channel(X, mean, std)
    mr = make_minirocket(Xn.shape[1])
    mr.fit(Xn)
    F = np.asarray(mr.transform(Xn), dtype=np.float32)   # (n_chunks, ~NUM_KERNELS)
    return mean, std, mr, F


def fit_lr(Ftr, ytr):
    """MiniRocket 특징(F) 위에서 StandardScaler + LogisticRegression 만 적합."""
    sc = StandardScaler().fit(Ftr)
    lr = LogisticRegression(max_iter=LR_MAX_ITER, class_weight=LR_CLASS_WEIGHT,
                        solver="saga", n_jobs=-1)
    lr.fit(sc.transform(Ftr), ytr)
    return sc, lr


def _proba_aligned(sc, lr, F):
    """(n, 5) — CLASS_ORDER 순서로 정렬된 확률(없는 클래스는 0). 이미 변환된 F 를 입력."""
    p = lr.predict_proba(sc.transform(F))
    full = np.zeros((len(F), len(CLASS_ORDER)), dtype=float)
    for j, c in enumerate(lr.classes_):
        full[:, CLASS_ORDER.index(str(c))] = p[:, j]
    return full


def slot_proba(model, X):
    """배포용: 원본 청크 X 를 받아 전체 파이프라인으로 5클래스 확률 반환."""
    Xn = _apply_channel(X, model["ch_mean"], model["ch_std"])
    F = np.asarray(model["mr"].transform(Xn), dtype=np.float32)
    return _proba_aligned(model["sc"], model["lr"], F)


def slot_predict(model, X):
    return np.array(CLASS_ORDER)[slot_proba(model, X).argmax(1)]


def train_slot(slot_key, X, y, g):
    """슬롯 1개 학습: 청크 캡 → MiniRocket 1회 변환 → StratifiedGroupKFold CV → 전체 재학습."""
    from sklearn.model_selection import StratifiedGroupKFold
    from sklearn.metrics import f1_score, accuracy_score, confusion_matrix

    info = dict(slot=slot_key, n_chunks=int(len(y)), classes=sorted(set(map(str, y))))
    if len(set(y)) < 2:
        info["status"] = "skipped: 단일 클래스 (분류 불가)"
        return None, info

    # (1) 슬롯·클래스당 청크 캡 (중복 제거)
    X, y, g = cap_per_class(X, y, g, CHUNK_CAP_PER_CLASS)
    info["n_chunks"] = int(len(y))
    info["classes"]  = sorted(set(map(str, y)))
    labels = sorted(set(map(str, y)))

    # (2) MiniRocket 변환은 슬롯당 1번만 → 폴드/최종학습에서 재사용
    #     주의: MiniRocket·채널표준화를 전체 데이터로 적합하므로 약한 정보 누수가 있으나,
    #     MiniRocket 은 비지도 변환이라 관례상 허용 수준(변환 비용 약 4배 절감 트레이드오프).
    mean, std, mr, F = build_features(X)

    # 클래스별 그룹 수로 가능한 분할 수 결정 (StratifiedGroupKFold 제약)
    min_groups = min(len(set(g[y == c])) for c in labels)
    n_splits = min(N_SPLITS, min_groups)

    if n_splits >= 2:
        sgkf = StratifiedGroupKFold(n_splits=n_splits, shuffle=True, random_state=RANDOM_STATE)
        f1s, accs, cm_total = [], [], None
        for tr, va in sgkf.split(np.zeros(len(y)), y, g):
            sc, lr = fit_lr(F[tr], y[tr])                       # ← 폴드마다 LR 만 재학습
            pred = np.array(CLASS_ORDER)[_proba_aligned(sc, lr, F[va]).argmax(1)]
            f1s.append(f1_score(y[va], pred, average="macro", labels=labels, zero_division=0))
            accs.append(accuracy_score(y[va], pred))
            cm = confusion_matrix(y[va], pred, labels=labels)
            cm_total = cm if cm_total is None else cm_total + cm
        info.update(status="ok", n_splits=int(n_splits), labels=labels,
                    macro_f1_mean=float(np.mean(f1s)), macro_f1_std=float(np.std(f1s)),
                    acc_mean=float(np.mean(accs)), acc_std=float(np.std(accs)),
                    confusion_matrix=cm_total.tolist())
    else:
        info.update(status=f"trained (CV 생략: 클래스당 그룹 수 부족 min_groups={min_groups})",
                    labels=labels)

    # (3) 최종 배포 모델: 동일 MiniRocket(mr) 재사용 + 전체 F 로 LR 재학습
    sc, lr = fit_lr(F, y)
    final_model = dict(ch_mean=mean, ch_std=std, mr=mr, sc=sc, lr=lr,
                       classes=list(map(str, lr.classes_)))
    return final_model, info

# =====================================================================
#  전체 학습 실행 → ModelRegistry + 평가 결과 저장
#   슬롯마다 중간 저장(체크포인트)하여 크래시/세션 종료에 대비합니다.
# =====================================================================

import json, joblib
from tqdm.auto import tqdm

if not MODEL_DIR:
    raise ValueError("MODEL_DIR(저장 경로)를 먼저 채워주세요.")
os.makedirs(MODEL_DIR, exist_ok=True)

reg_path  = os.path.join(MODEL_DIR, "model_registry.joblib")
eval_path = os.path.join(MODEL_DIR, "eval_results.json")

registry, eval_rows = {}, []
for slot_key in tqdm(sorted(slots, key=lambda x: (str(x[0]), str(x[1]))), desc="슬롯 학습"):
    data = load_slot_chunks(slots[slot_key])
    if data is None:
        eval_rows.append(dict(slot=slot_key, n_chunks=0, status="skipped: 청크 없음"))
        continue
    X, y, g = data
    model, info = train_slot(slot_key, X, y, g)
    eval_rows.append(info)
    if model is not None:
        registry[slot_key] = model
        joblib.dump(registry, reg_path)                 # ← 슬롯마다 중간 저장
        with open(eval_path, "w", encoding="utf-8") as f:
            json.dump(eval_rows, f, ensure_ascii=False, indent=2, default=str)

# 최종 저장 (루프에서 이미 저장되지만 안전하게 한 번 더)
joblib.dump(registry, reg_path)
with open(eval_path, "w", encoding="utf-8") as f:
    json.dump(eval_rows, f, ensure_ascii=False, indent=2, default=str)

print(f"\n학습 완료 — 등록된 모델 {len(registry)}개")
print("저장:", reg_path)

import joblib, numpy as np, os, time

def balanced_sample_idx(y, k, seed=42):
    # 클래스가 골고루 섞이도록 k개 인덱스를 round-robin 으로 뽑는다.
    rng = np.random.default_rng(seed)
    y = np.asarray(y)
    classes = np.unique(y)

    by_class = {
        c: list(rng.permutation(np.where(y == c)[0]))
        for c in classes
    }

    ptr = {c: 0 for c in classes}
    picked = []

    while len(picked) < k:
        progressed = False

        for c in classes:
            if ptr[c] < len(by_class[c]):
                picked.append(int(by_class[c][ptr[c]]))
                ptr[c] += 1
                progressed = True

                if len(picked) >= k:
                    break

        if not progressed:
            break

    return picked

셀 1 — fit_lr 솔버 교체 + build_features F 캐시¶

  • 솔버: saga → lbfgs. ConvergenceWarning 의 진짜 원인은 saga 의 느린 수렴이었다 (StandardScaler 는 fit_lr 에 이미 있었음). lbfgs 는 L2 로지스틱·고차원에서 빠르게 수렴.
  • F 캐시: build_features(X) 결과를 X 내용 해시로 .npz 저장. 솔버·C·max_iter 만 바꿔 재학습할 때 무거운 MiniRocket 변환을 건너뛴다. (cap_per_class 가 seed 고정이라 같은 슬롯이면 X 가 동일 → 캐시 히트)
  • 병렬 대비: MiniRocket 내부 병렬을 1로 낮춘다. 바깥에서 슬롯 3개를 병렬로 돌리므로 안쪽까지 전체 코어를 쓰면 oversubscription 으로 오히려 느려진다.

(D) 12채널 융합 A/B 비교 — 선택 실험¶

왜 하나? 현재 센서 분리(late fusion: 센서별 모델 → 확률을 0.6평균+0.4최고 로 합침)는 센서 간 결합 정보와 적응형 가중치를 포기한 트레이드오프다. 어떤 고장은 전류 패턴 + 진동 패턴이 같이 나타나는 조합으로만 구별되는데, 따로 예측해 나중에 합치면 그 결합을 못 쓴다. 또 고정 가중치(0.6/0.4)는 고장마다 어떤 센서가 중요한지 자동으로 못 바꾼다.

대안: 한 세션의 3센서를 채널로 쌓아(3센서 × 4피처 = 12채널) 한 모델에 한 샘플로 넣는다. MiniRocketMultivariate 는 채널 수와 무관하게 받으므로 코드 변경이 작다. 이러면 한 모델이 센서를 동시에 보고 결합 정보를 학습한다.

대전제: 3센서가 시간 정렬돼 있어야 채널로 쌓을 수 있다. 그래서 먼저 정렬 점검부터 한다 — 짝이 충분치 않으면 융합 샘플을 못 만든다.

def load_slot_chunks(paths):
    """
    슬롯에 속한 CSV 들을 읽어 청크 단위 샘플로 변환.
    반환: X (n_chunks, C, CHUNK_SIZE), y (n_chunks,), groups (n_chunks,) | None
    각 청크의 label = 해당 CSV 의 fault_class, group = group_key (누수 방지용).
    """
    Xs, ys, gs = [], [], []
    for p in paths:
        try:
            df = pd.read_csv(p)
        except Exception as e:
            print("스킵(읽기 실패):", p, "→", e); continue
        if df.empty:
            continue
        feats = df[FEATURE_COLS].to_numpy(dtype=np.float32)     # (n, C)
        n = (len(feats) // CHUNK_SIZE) * CHUNK_SIZE             # 끝의 잔여행(<2000) 버림
        if n == 0:
            continue
        C = feats.shape[1]
        chunks = feats[:n].T.reshape(C, -1, CHUNK_SIZE).transpose(1, 0, 2)  # (n_chunks, C, L)
        label = str(df["fault_class"].iloc[0])
        group = str(df["group_key"].iloc[0])
        Xs.append(chunks)
        ys += [label] * len(chunks)
        gs += [group] * len(chunks)
    if not Xs:
        return None
    return np.concatenate(Xs, 0), np.array(ys), np.array(gs)

import numpy as np

def _chunkify(feats, chunk_size):                 # feats: (n, C) -> (n_chunks, C, L)
    n = (len(feats) // chunk_size) * chunk_size
    if n == 0:
        return None
    C = feats.shape[1]
    return feats[:n].T.reshape(C, -1, chunk_size).transpose(1, 0, 2)

def build_vehicle_fusion(vehicle, sessions):
    # 차종 1대의 모든 완비 세션 -> (X12, y, groups). X12: (N, 12, CHUNK_SIZE)
    Xs, ys, gs = [], [], []
    for sess in sessions:
        smap = session_index[(vehicle, sess)]
        # 센서별 4피처 로드
        feats_by_sensor = {}
        ok = True
        for s in SENSORS:
            try:
                d = pd.read_csv(smap[s]).sort_values("time_idx")
                feats_by_sensor[s] = d[FEATURE_COLS].to_numpy(dtype=np.float32)  # (T,4)
            except Exception as e:
                print("  [스킵]", vehicle, sess, s, "->", e); ok = False; break
        if not ok:
            continue
        T = min(f.shape[0] for f in feats_by_sensor.values())          # 동일 길이면 그대로
        stacked = np.concatenate([feats_by_sensor[s][:T] for s in SENSORS], axis=1)  # (T,12)
        ch = _chunkify(stacked, CHUNK_SIZE)
        if ch is None:
            continue
        Xs.append(ch)
        ys += [session_label[(vehicle, sess)]] * len(ch)
        gs += [sess] * len(ch)                                          # group=세션
    if not Xs:
        return None
    return np.concatenate(Xs, 0), np.array(ys), np.array(gs)

import json, os, numpy as np, pandas as pd

CLASS_ORDER = ["DEMAG", "ECC10", "ECC20", "NORMAL", "REDUC"]

def per_class_f1(cm, labels):
    cm = np.array(cm, dtype=float)
    out = {}
    for i, c in enumerate(labels):
        tp = cm[i, i]
        fp = cm[:, i].sum() - tp
        fn = cm[i, :].sum() - tp
        prec = tp/(tp+fp) if (tp+fp) > 0 else 0.0
        rec  = tp/(tp+fn) if (tp+fn) > 0 else 0.0
        out[c] = round(2*prec*rec/(prec+rec), 3) if (prec+rec) > 0 else 0.0
    return out

def load_eval(fname):
